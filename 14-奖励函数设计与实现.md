## 14.1 奖励函数在 verl 中的实现

### 14.1.1 基本奖励函数接口

```python
def compute_single_reward(completion, ground_truth):
    """单条回复的奖励计算"""
    if ground_truth in completion:
        return 1.0
    return 0.0

def reward_fn(completions: list, ground_truths: list, **kwargs) -> list:
    rewards = []
    for completion, gt in zip(completions, ground_truths):
        reward = compute_single_reward(completion, gt)
        rewards.append(reward)
    return rewards
```

### 14.1.2 多组件奖励函数

```python
import re

def extract_final_answer(completion):
    """从回复中提取最终答案"""
    match = re.search(r'(?:答案[是为：:\s]*|answer[:\s]*)(.+)', completion, re.I)
    return match.group(1).strip() if match else completion.strip()

def has_valid_json_tool_call(text):
    """检查是否包含合法的 JSON 工具调用"""
    return bool(re.search(r'\{\s*["\']name["\']\s*:\s*["\']\w+["\']', text))

def has_repetition(text, threshold=3):
    """检查是否存在重复片段"""
    lines = [l.strip() for l in text.split('\n') if l.strip()]
    if len(lines) < threshold:
        return False
    recent = lines[-threshold:]
    return len(set(recent)) == 1

class MultiComponentReward:
    def __init__(self):
        self.components = [
            ("correctness", 1.0, self.correctness_reward),
            ("format", 0.1, self.format_reward),
            ("efficiency", 0.05, self.efficiency_reward),
            ("no_hacking", -0.5, self.anti_hacking_penalty),
        ]
    
    def __call__(self, completions, ground_truths, trajectories=None):
        rewards = []
        for _, (comp, gt) in enumerate(zip(completions, ground_truths)):
            traj = trajectories[_] if trajectories else None
            total = 0.0
            for _, weight, fn in self.components:
                total += weight * fn(comp, gt, traj)
            rewards.append(total)
        return rewards
    
    def correctness_reward(self, completion, gt, traj=None):
        answer = extract_final_answer(completion)
        return 1.0 if answer == gt else 0.0
    
    def format_reward(self, completion, gt, traj=None):
        score = 0.0
        if '<think>' in completion or '</think>' in completion:
            score += 0.5
        if has_valid_json_tool_call(completion):
            score += 0.5
        return score
    
    def efficiency_reward(self, completion, gt, traj):
        if traj:
            steps = len(traj.get("steps", []))
            return max(0, 1 - steps / 10)
        return 0.0
    
    def anti_hacking_penalty(self, completion, gt, traj):
        if has_repetition(completion):
            return 1.0
        return 0.0
```

## 14.2 不同场景的奖励设计

### 14.2.1 数学推理奖励

```python
import re

def normalize_math(expr):
    """标准化数学表达式用于比较"""
    expr = re.sub(r'\s+', '', expr)
    expr = re.sub(r'[\\,;\.]$', '', expr)
    return expr.lower()

def math_reward(completion, ground_truth):
    """数学推理的奖励函数"""
    reward = 0.0
    
    answer = None
    match = re.search(r'\\boxed\{(.+?)\}', completion)
    if match:
        answer = match.group(1).strip()
    if not answer:
        match = re.search(r'(?:答案|answer)[是为：:\s]+(.+)', completion, re.I)
        if match:
            answer = match.group(1).strip()
    
    if answer and normalize_math(answer) == normalize_math(str(ground_truth)):
        reward += 1.0
    
    if '<think>' in completion or '</think>' in completion or "step" in completion.lower():
        reward += 0.05
    
    return reward
```

### 14.2.2 代码生成奖励

```python
import ast

def extract_code_block(completion):
    """从回复中提取代码块"""
    match = re.search(r'```(?:python)?\n(.*?)```', completion, re.DOTALL)
    if match:
        return match.group(1).strip()
    return ""

def safe_execute(code, test_input):
    """安全执行代码并返回结果"""
    namespace = {}
    try:
        exec(code, namespace)
        if "solution" in namespace:
            return namespace["solution"](test_input)
        if "solve" in namespace:
            return namespace["solve"](test_input)
    except Exception:
        return None

def code_reward(completion, test_cases):
    """代码生成的奖励函数"""
    code = extract_code_block(completion)
    if not code:
        return 0.0
    
    passed = 0
    total = len(test_cases)
    
    if total == 0:
        return 0.0
    
    for test_input, expected in test_cases:
        try:
            result = safe_execute(code, test_input)
            if result == expected:
                passed += 1
        except:
            pass
    
    reward = passed / total
    
    if passed == total:
        reward += 0.2
    
    return min(reward, 1.0)
```

### 14.2.3 Web Agent 奖励

```python
def task_completed(trajectory, task_config):
    """判断任务是否完成"""
    final_state = trajectory.get("final_state", {})
    goal = task_config.get("goal", "")
    return goal in str(final_state)

def count_invalid_actions(trajectory):
    """统计无效操作次数"""
    actions = trajectory.get("actions", [])
    return sum(1 for a in actions if a.get("error") or a.get("success") is False)

def web_agent_reward(trajectory, task_config):
    """网页导航 Agent 的奖励函数"""
    reward = 0.0
    
    if task_completed(trajectory, task_config):
        reward += 1.0
    
    num_steps = len(trajectory.get("actions", []))
    max_steps = task_config.get("max_steps", 15)
    if num_steps <= max_steps:
        reward += 0.1 * (1 - num_steps / max_steps)
    
    invalid_actions = count_invalid_actions(trajectory)
    reward -= 0.05 * invalid_actions
    
    return max(reward, -0.5)
```

### 14.2.4 SQL Agent 奖励

```python
def extract_sql(completion):
    """从回复中提取 SQL 查询语句"""
    match = re.search(r'```sql\n(.*?)```', completion, re.DOTALL)
    if match:
        return match.group(1).strip()
    lines = [l for l in completion.split('\n') if l.strip().upper().startswith(('SELECT', 'INSERT', 'UPDATE', 'DELETE'))]
    return lines[0] if lines else ""

def sql_agent_reward(completion, expected_result, db_connection):
    """SQL Agent 的奖励函数"""
    sql_query = extract_sql(completion)
    if not sql_query:
        return 0.0
    
    try:
        cursor = db_connection.cursor()
        cursor.execute(sql_query)
        if cursor.description:
            actual_rows = cursor.fetchall()
            actual_columns = [d[0] for d in cursor.description]
            actual_result = {"rows": actual_rows, "columns": actual_columns}
        else:
            db_connection.commit()
            return 0.5 if cursor.rowcount > 0 else 0.0
    except Exception:
        return -0.1
    
    if actual_result == expected_result:
        return 1.0
    
    score = 0.0
    if actual_result["columns"] == expected_result.get("columns", []):
        score += 0.3
    if len(actual_result["rows"]) == len(expected_result.get("rows", [])):
        score += 0.2
    
    return score
```

## 14.3 奖励调试技巧

### 14.3.1 奖励分布监控

```python
import numpy as np

def analyze_reward_distribution(rewards_batch):
    """分析奖励分布"""
    rewards = np.array(rewards_batch)
    print(f"Mean: {rewards.mean():.3f}")
    print(f"Std:  {rewards.std():.3f}")
    print(f"Min:  {rewards.min():.3f}")
    print(f"Max:  {rewards.max():.3f}")
    print(f"All same: {np.all(rewards == rewards[0])}")
    # 如果所有奖励相同，GRPO 的梯度为0，训练停滞！
```

### 14.3.2 常见奖励问题诊断

| 问题 | 症状 | 解决方案 |
|------|------|---------|
| 奖励全相同 | GRPO 梯度为0，训练停滞 | 增加任务难度多样性 |
| 奖励太极端 | 训练不稳定 | 截断（clip）奖励范围 |
| 奖励太稀疏 | 训练收敛慢 | 增加中间奖励信号 |
| 奖励被Hack | 模型得分高但能力差 | 增加反作弊检查 |

## 14.5 小结

- 奖励函数设计是 Agentic RL 最核心的工程环节
- 多组件奖励可以平衡正确性、格式、效率和安全性
- 不同场景需要定制不同的奖励函数
- 监控奖励分布，防止训练停滞
