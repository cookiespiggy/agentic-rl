## 13.1 RL 环境的核心接口

无论什么类型的环境，都需要实现以下标准接口：

```python
from abc import ABC, abstractmethod

class BaseEnvironment(ABC):
    """RL 环境的基类（Base Class）"""
    
    @abstractmethod
    def reset(self) -> dict:
        """重置环境，返回初始观测"""
        pass
    
    @abstractmethod
    def step(self, action: str) -> tuple:
        """执行动作，返回 (观测, 奖励, 是否结束, 额外信息)"""
        pass
    
    @abstractmethod
    def get_observation(self) -> str:
        """获取当前观测"""
        pass
    
    @abstractmethod
    def compute_reward(self) -> float:
        """计算奖励"""
        pass
```

## 13.2 代码执行环境（Code Sandbox）

### 13.2.1 简单的 Python 代码执行

```python
import os
import subprocess
import tempfile

class CodeSandbox:
    """安全的 Python 代码执行沙盒"""
    
    def __init__(self, timeout=10):
        self.timeout = timeout
    
    def execute(self, code: str) -> dict:
        """执行 Python 代码"""
        tmp = tempfile.NamedTemporaryFile(mode='w', suffix='.py', delete=False)
        try:
            tmp.write(code)
            tmp.flush()
            tmp.close()
            
            result = subprocess.run(
                ['python', tmp.name],
                capture_output=True,
                text=True,
                timeout=self.timeout
            )
            return {
                "stdout": result.stdout,
                "stderr": result.stderr,
                "return_code": result.returncode,
                "success": result.returncode == 0
            }
        except subprocess.TimeoutExpired:
            return {"stdout": "", "stderr": "Timeout", "success": False}
        finally:
            os.unlink(tmp.name)

sandbox = CodeSandbox()
result = sandbox.execute("print(2 + 3)")
print(result)
```

## 13.3 搜索环境（Search Environment）

```python
import requests

class SearchEnvironment:
    """搜索引擎交互环境"""
    
    def __init__(self, search_api_key, max_results=5):
        self.search_api_key = search_api_key
        self.max_results = max_results
        self.search_history = []
    
    def search(self, query: str) -> list:
        """执行搜索（以 SerpAPI 为例）"""
        response = requests.get(
            "https://serpapi.com/search",
            params={"q": query, "api_key": self.search_api_key, "num": self.max_results}
        )
        results = response.json().get("organic_results", [])
        self.search_history.append({"query": query, "results": results})
        return results
    
    def compute_reward(self, final_answer, ground_truth):
        """基于最终答案计算奖励"""
        if final_answer.strip() == ground_truth.strip():
            return 1.0
        return 0.0
```

## 13.5 数据库查询环境

```python
import sqlite3

class SQLEnvironment:
    """SQL 查询交互环境"""
    
    def __init__(self, db_path, schema_info):
        self.db_path = db_path
        self.conn = sqlite3.connect(db_path)
        self.schema_info = schema_info
    
    def __enter__(self):
        return self
    
    def __exit__(self, *args):
        self.close()
    
    def close(self):
        if self.conn:
            self.conn.close()
            self.conn = None
    
    def execute_sql(self, query: str) -> dict:
        """执行 SQL 查询"""
        try:
            cursor = self.conn.cursor()
            cursor.execute(query)
            if cursor.description:
                results = cursor.fetchall()
                columns = [desc[0] for desc in cursor.description]
                return {
                    "success": True,
                    "results": results,
                    "columns": columns,
                    "row_count": len(results)
                }
            self.conn.commit()
            return {"success": True, "results": [], "columns": [], "row_count": cursor.rowcount}
        except Exception as e:
            return {"success": False, "error": str(e)}
    
    def compute_reward(self, query, expected_result):
        """对比查询结果"""
        actual = self.execute_sql(query)
        if not actual["success"]:
            return 0.0
        if actual["results"] == expected_result:
            return 1.0
        if actual["columns"] and expected_result and "columns" in expected_result:
            if set(actual["columns"]) == set(expected_result["columns"]):
                return 0.5
        return 0.0
```

## 13.6 verl 中的环境配置

### 13.6.1 定义工具配置

```yaml
# tools_config.yaml
tools:
  - name: "python_interpreter"
    description: "Execute Python code"
    parameters:
      code:
        type: "string"
        description: "Python code to execute"
    handler: "code_sandbox.execute"
  
  - name: "search"
    description: "Search the web"
    parameters:
      query:
        type: "string"
        description: "Search query"
    handler: "search_engine.search"
```

### 13.6.2 verl 工具注册

```python
import os
from verl.tools import ToolRegistry

# 注册工具
@ToolRegistry.register("code_interpreter")
def code_interpreter(code: str):
    """代码解释器工具"""
    sandbox = CodeSandbox(timeout=10)
    return sandbox.execute(code)

@ToolRegistry.register("search")
def search(query: str):
    """搜索工具"""
    engine = SearchEnvironment(search_api_key=os.environ["SEARCH_API_KEY"])
    return engine.search(query)
```
